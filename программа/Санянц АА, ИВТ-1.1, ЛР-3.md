## Задача 1.2: Операции над 3D-векторами

### Постановка задачи:  
Создать структуру для вектора в 3-х мерном пространстве и реализовать операции:
- Скалярное умножение векторов
- Векторное произведение
- Вычисление модуля вектора
- Вывод вектора в консоль

Структура должна содержать имя вектора как отдельное поле.

### Математическая модель:  

\[
\vec{a} \cdot \vec{b} = x_1 x_2 + y_1 y_2 + z_1 z_2
\]

\[
\vec{a} \times \vec{b} = (y_1 z_2 - z_1 y_2,\; z_1 x_2 - x_1 z_2,\; x_1 y_2 - y_1 x_2)
\]

\[
|\vec{a}| = \sqrt{x_1^2 + y_1^2 + z_1^2}
\]

### Список идентификаторов:  
| Идентификатор | Тип данных | Описание |
|---------------|------------|----------|
| `Vector3D` | `struct` | Структура для 3D-вектора |
| `name` | `char[50]` | Имя вектора |
| `x, y, z` | `double` | Координаты |
| `dot_product` | `double` | Скалярное произведение |
| `cross_product` | `Vector3D` | Векторное произведение |
| `vector_length` | `double` | Длина вектора |

### Код программы:
```c
#include <stdio.h>
#include <math.h>
#include <string.h>

typedef struct {
    char name[50];
    double x, y, z;
} Vector3D;

double dot_product(Vector3D a, Vector3D b) {
    return a.x * b.x + a.y * b.y + a.z * b.z;
}

Vector3D cross_product(Vector3D a, Vector3D b) {
    Vector3D result;
    strcpy(result.name, "Векторное произведение");
    result.x = a.y * b.z - a.z * b.y;
    result.y = a.z * b.x - a.x * b.z;
    result.z = a.x * b.y - a.y * b.x;
    return result;
}

double vector_length(Vector3D v) {
    return sqrt(v.x * v.x + v.y * v.y + v.z * v.z);
}

void print_vector(Vector3D v) {
    printf("%s: (%.2f, %.2f, %.2f)\n", v.name, v.x, v.y, v.z);
}

int main() {
    Vector3D v1 = {"Вектор A", 1.0, 2.0, 3.0};
    Vector3D v2 = {"Вектор B", 4.0, 5.0, 6.0};
    
    print_vector(v1);
    print_vector(v2);
    
    double dot = dot_product(v1, v2);
    printf("Скалярное произведение: %.2f\n", dot);
    
    Vector3D cross = cross_product(v1, v2);
    print_vector(cross);
    
    printf("Длина вектора %s: %.2f\n", v1.name, vector_length(v1));
    printf("Длина вектора %s: %.2f\n", v2.name, vector_length(v2));
    
    return 0;
}
```

## Задача 1.3: Комплексная экспонента

### Постановка задачи:  
Вычислить комплексную экспоненту \( \exp(z) \) для \( z \in \mathbb{C} \):

\[
\exp(z) = 1 + z + \frac{1}{2!}z^2 + \frac{1}{3!}z^3 + \ldots + \frac{1}{n!}z^n
\]

### Математическая модель:  

\[
e^{a+bi} = e^a (\cos b + i \sin b)
\]

Рекуррентное соотношение для членов ряда:

\[
\frac{z^k}{k!} = \frac{z^{k-1}}{(k-1)!} \cdot \frac{z}{k}
\]

### Список идентификаторов:  
| Идентификатор | Тип данных | Описание |
|---------------|------------|----------|
| `Complex` | `struct` | Комплексное число |
| `real, imag` | `double` | Действительная и мнимая части |
| `complex_exp` | `Complex` | Функция вычисления экспоненты |

### Код программы:
```c
#include <stdio.h>
#include <math.h>

typedef struct {
    double real;
    double imag;
} Complex;

Complex add(Complex a, Complex b) {
    Complex result;
    result.real = a.real + b.real;
    result.imag = a.imag + b.imag;
    return result;
}

Complex multiply(Complex a, Complex b) {
    Complex result;
    result.real = a.real * b.real - a.imag * b.imag;
    result.imag = a.real * b.imag + a.imag * b.real;
    return result;
}

Complex divide_by_scalar(Complex a, double scalar) {
    Complex result;
    result.real = a.real / scalar;
    result.imag = a.imag / scalar;
    return result;
}

Complex complex_exp(Complex z, int terms) {
    Complex result = {1.0, 0.0};
    Complex current_term = {1.0, 0.0};
    
    for (int n = 1; n <= terms; n++) {
        current_term = multiply(current_term, z);
        current_term = divide_by_scalar(current_term, n);
        result = add(result, current_term);
    }
    
    return result;
}

void print_complex(Complex z) {
    if (z.imag >= 0)
        printf("%.4f + %.4fi", z.real, z.imag);
    else
        printf("%.4f - %.4fi", z.real, -z.imag);
}

int main() {
    Complex z;
    int terms;
    
    printf("Введите действительную часть: ");
    scanf("%lf", &z.real);
    printf("Введите мнимую часть: ");
    scanf("%lf", &z.imag);
    printf("Введите количество членов ряда: ");
    scanf("%d", &terms);
    
    Complex exp_z = complex_exp(z, terms);
    
    printf("\nexp(");
    print_complex(z);
    printf(") = ");
    print_complex(exp_z);
    printf("\n");
    
    return 0;
}
```

## Задача 1.4: Битовые поля для даты

### Постановка задачи:  
Используя битовые поля, создать экономичную структуру для хранения даты.

### Математическая модель:  
Не требуется.

### Список идентификаторов:  
| Идентификатор | Тип данных | Описание |
|---------------|------------|----------|
| `CompactDate` | `struct` | Дата с битовыми полями |
| `day` | `unsigned int:5` | День (1-31) |
| `month` | `unsigned int:4` | Месяц (1-12) |
| `year` | `unsigned int:12` | Год (0-4095) |

### Код программы:
```c
#include <stdio.h>

typedef struct {
    unsigned int day : 5;
    unsigned int month : 4;
    unsigned int year : 12;
} CompactDate;

void print_date(const CompactDate* date) {
    printf("Дата: %02u.%02u.%04u\n", date->day, date->month, date->year);
}

int is_date_valid(const CompactDate* date) {
    if (date->month == 0 || date->month > 12) return 0;
    if (date->day == 0) return 0;
    
    int days_in_month;
    switch (date->month) {
        case 2:
            days_in_month = ((date->year % 4 == 0 && date->year % 100 != 0) || date->year % 400 == 0) ? 29 : 28;
            break;
        case 4: case 6: case 9: case 11:
            days_in_month = 30;
            break;
        default:
            days_in_month = 31;
    }
    
    return date->day <= days_in_month;
}

int main() {
    CompactDate birthday = {15, 6, 1990};
    
    printf("Размер структуры: %zu байт\n", sizeof(CompactDate));
    print_date(&birthday);
    printf("Дата %s\n", is_date_valid(&birthday) ? "корректна" : "некорректна");
    
    CompactDate invalid = {31, 2, 2023};
    print_date(&invalid);
    printf("Дата %s\n", is_date_valid(&invalid) ? "корректна" : "некорректна");
    
    return 0;
}
```

## Задача 1.5: Двунаправленный связный список

### Постановка задачи:  
Реализовать двунаправленный связный список и выполнить обход в прямом и обратном направлениях.

### Математическая модель:  
Не требуется.

### Список идентификаторов:  
| Идентификатор | Тип данных | Описание |
|---------------|------------|----------|
| `Node` | `struct` | Элемент списка |
| `value` | `int` | Значение |
| `prev, next` | `struct Node*` | Указатели |

### Код программы:
```c
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int value;
    struct Node* prev;
    struct Node* next;
};

struct Node* create(int val) {
    struct Node* n = malloc(sizeof(struct Node));
    n->value = val;
    n->prev = n->next = NULL;
    return n;
}

void add(struct Node** head, struct Node** tail, int val) {
    struct Node* n = create(val);
    
    if (*head == NULL) {
        *head = *tail = n;
    } else {
        (*tail)->next = n;
        n->prev = *tail;
        *tail = n;
    }
}

void print_forward(struct Node* head) {
    printf("Вперед: ");
    while (head) {
        printf("%d ", head->value);
        head = head->next;
    }
    printf("\n");
}

void print_backward(struct Node* tail) {
    printf("Назад: ");
    while (tail) {
        printf("%d ", tail->value);
        tail = tail->prev;
    }
    printf("\n");
}

void clear(struct Node* head) {
    while (head) {
        struct Node* t = head;
        head = head->next;
        free(t);
    }
}

int main() {
    struct Node *head = NULL, *tail = NULL;
    
    add(&head, &tail, 10);
    add(&head, &tail, 20);
    add(&head, &tail, 30);
    
    print_forward(head);
    print_backward(tail);
    
    clear(head);
    return 0;
}
```

## Задача 2.4: Размеченное объединение

### Постановка задачи:  
Создать размеченное объединение (struct + union + enum), динамический массив таких структур и распечатку.

### Код программы:
```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

enum DataType { INT, FLOAT, STRING };

typedef struct {
    enum DataType type;
    union {
        int i;
        float f;
        char s[20];
    } data;
} TaggedUnion;

void print_tagged_union(const TaggedUnion* tu) {
    switch (tu->type) {
        case INT:
            printf("Тип: INT, значение: %d\n", tu->data.i);
            break;
        case FLOAT:
            printf("Тип: FLOAT, значение: %.2f\n", tu->data.f);
            break;
        case STRING:
            printf("Тип: STRING, значение: %s\n", tu->data.s);
            break;
    }
}

int main() {
    TaggedUnion* array = malloc(3 * sizeof(TaggedUnion));
    
    array[0].type = INT;
    array[0].data.i = 42;
    
    array[1].type = FLOAT;
    array[1].data.f = 3.14159f;
    
    array[2].type = STRING;
    strcpy(array[2].data.s, "Hello, World!");
    
    printf("Содержимое массива:\n");
    for (int i = 0; i < 3; i++) {
        print_tagged_union(&array[i]);
    }
    
    free(array);
    return 0;
}
```
